TABLA: ubicaciones
- id: bigint, PK, autoincremental
- nombre: varchar
- tipo: enum('oficina', 'bodega', 'vehiculo', 'via_publica')
- direccion: varchar, nullable
- created_at: timestamp
- updated_at: timestamp

TABLA: tipos
- id: bigint, PK, autoincremental
- nombre: varchar, unique
- requiere_serie: boolean
- controla_vencimiento: boolean
- created_at: timestamp
- updated_at: timestamp

TABLA: funcionarios
- id: bigint, PK, autoincremental
- rut: varchar, unique
- nombres: varchar
- apellidos: varchar
- cargo: varchar, nullable
- activo: boolean
- created_at: timestamp
- updated_at: timestamp

TABLA: items
- id: bigint, PK, autoincremental
- numero_inventario: varchar, unique, nullable
- codigo_antiguo: varchar, nullable
- tipo_id: bigint, FK -> tipos.id
- ubicacion_id: bigint, FK -> ubicaciones.id, nullable
- funcionario_id: bigint, FK -> funcionarios.id, nullable
- item_padre_id: bigint, FK -> items.id, nullable
- marca: varchar, nullable
- modelo: varchar, nullable
- numero_serie: varchar, nullable
- descripcion: text, nullable
- fecha_vencimiento: date, nullable
- estado_conservacion: enum('bueno', 'regular', 'malo')
- situacion: enum('registrado_daf', 'sin_registro_daf', 'solicitud_baja', 'retirado')
- created_at: timestamp
- updated_at: timestamp
- deleted_at: timestamp, nullable (SoftDeletes)

TABLA: users
- id: bigint, PK, autoincremental
- name: varchar
- email: varchar, unique
- email_verified_at: timestamp, nullable
- password: varchar
- remember_token: varchar, nullable
- created_at: timestamp
- updated_at: timestamp

TABLA: procesos_verificacion
- id: bigint, PK, autoincremental
- nombre: varchar
- fecha_inicio: date
- fecha_cierre: date, nullable
- estado: enum('abierto', 'cerrado')
- user_id: bigint, FK -> users.id
- created_at: timestamp
- updated_at: timestamp

TABLA: verificaciones
- id: bigint, PK, autoincremental
- proceso_verificacion_id: bigint, FK -> procesos_verificacion.id
- item_id: bigint, FK -> items.id, nullable
- ubicacion_id: bigint, FK -> ubicaciones.id, nullable
- funcionario_id: bigint, FK -> funcionarios.id, nullable
- resultado: enum('encontrado', 'no_encontrado', 'sin_registro')
- metodo: enum('fisica', 'declarada')
- estado_conservacion: enum('bueno', 'regular', 'malo'), nullable
- codigo_observado: varchar, nullable
- descripcion_observada: text, nullable
- observacion: text, nullable
- user_id: bigint, FK -> users.id
- verificado_at: timestamp
- created_at: timestamp
- updated_at: timestamp

TABLA: roles (Spatie laravel-permission)
- id: bigint, PK, autoincremental
- name: varchar
- guard_name: varchar
- created_at: timestamp
- updated_at: timestamp
- unique(name, guard_name)

TABLA: permissions (Spatie laravel-permission)
- id: bigint, PK, autoincremental
- name: varchar
- guard_name: varchar
- created_at: timestamp
- updated_at: timestamp
- unique(name, guard_name)

TABLA: model_has_roles (Spatie laravel-permission)
- role_id: bigint, FK -> roles.id
- model_type: varchar
- model_id: bigint
- PK compuesta (role_id, model_id, model_type)

TABLA: model_has_permissions (Spatie laravel-permission)
- permission_id: bigint, FK -> permissions.id
- model_type: varchar
- model_id: bigint
- PK compuesta (permission_id, model_id, model_type)

TABLA: role_has_permissions (Spatie laravel-permission)
- permission_id: bigint, FK -> permissions.id
- role_id: bigint, FK -> roles.id
- PK compuesta (permission_id, role_id)

TABLA: activity_log (Spatie laravel-activitylog)
- id: bigint, PK, autoincremental
- log_name: varchar, nullable
- description: text
- subject_type: varchar, nullable
- subject_id: bigint, nullable
- event: varchar, nullable
- causer_type: varchar, nullable
- causer_id: bigint, nullable
- properties: json, nullable
- batch_uuid: uuid, nullable
- created_at: timestamp
- updated_at: timestamp
